
class NikoWorld
  attr_accessor :position, :light_on, :inventory, :world_state, :memory_flags

  WORLDS = [:cottage, :abandoned_fields, :factory, :refuge, :tower_entrance]

  def initialize
    @position = {x: 0, y: 0}
    @light_on = false
    @inventory = []
    @world_state = :cottage
    @memory_flags = {}

    log "WORLD_INIT >> World: #{@world_state} Position: #{@position}"
  end

  # -----------------------------
  # MOVEMENT + ENERGY SIMULATION
  # -----------------------------
  def move(dx, dy)
    @position[:x] += dx
    @position[:y] += dy

    log_position
    ambient_events
  end

  # -----------------------------
  # ITEM PICKUP
  # -----------------------------
  def take(item)
    @inventory << item
    log "INVENTORY >> Picked up #{item}"

    case item
    when :sun
      @light_on = true
      log "LIGHT >> Sun is glowing warmly"
    when :crowbar
      log "TOOL >> Can pry open loose panels"
    when :journal
      log "MEMORY >> Something feels written between the lines"
    end
  end

  # -----------------------------
  # INTERACTION LOGIC
  # -----------------------------
  def interact(target)
    case target
    when :bed
      log "REST >> Niko doesn’t feel sleepy… not yet."
    when :computer
      computer_sequence
    when :robot
      robot_interaction
    when :door
      try_enter_next_world
    when :panel
      panel_attempt
    else
      log "INTERACT >> Niko doesn't understand this object."
    end
  end

  # -----------------------------
  # COMPUTER EVENT
  # (OneShot-themed but not copying anything)
  # -----------------------------
  def computer_sequence
    if @memory_flags[:factory_power]
      log "TERMINAL >> 'Systems restored. Proceed.'"
    else
      log "TERMINAL >> 'The world is fragile.'"
      @memory_flags[:computer_seen] = true
    end
  end

  # -----------------------------
  # ROBOT INTERACTION
  # -----------------------------
  def robot_interaction
    if @light_on
      log "ROBOT >> 'The Sun… you are carrying it… thank you.'"
    else
      log "ROBOT >> 'Unit perception limited. No light detected.'"
    end
  end

  # -----------------------------
  # WORLD TRANSITION
  # -----------------------------
  def try_enter_next_world
    index = WORLDS.index(@world_state)

    if @light_on
      if index < WORLDS.size - 1
        @world_state = WORLDS[index + 1]
        @position = {x: 0, y: 0}
        log "WORLD_SHIFT >> Entered #{@world_state}"
      else
        log "TOWER >> The rest is up to Niko."
      end
    else
      log "DOOR >> It won't budge in the darkness."
    end
  end

  # -----------------------------
  # PANEL / FACTORY PUZZLE
  # -----------------------------
  def panel_attempt
    if @inventory.include?(:crowbar)
      log "PANEL >> Forced open. Exposed wiring."
      @memory_flags[:factory_power] = true
    else
      log "PANEL >> Bolted shut. Needs something sturdy."
    end
  end

  # -----------------------------
  # WORLD EVENTS / ATMOSPHERE
  # -----------------------------
  def ambient_events
    case @world_state
    when :cottage
      log "AMBIENCE >> Dusty wood creaks softly." if rand < 0.1
    when :abandoned_fields
      log "AMBIENCE >> The phosphor grass pulses faintly." if @light_on
    when :factory
      log "AMBIENCE >> Distant machinery hums." if @memory_flags[:factory_power]
    when :refuge
      log "AMBIENCE >> The air feels uneasy." if rand < 0.05
    end
  end

  # -----------------------------
  # UTILITY LOGGING
  # -----------------------------
  def log(str)
    puts str
  end

  def log_position
    log "POS >> #{@world_state} X:#{@position[:x]} Y:#{@position[:y]} Light:#{@light_on}"
  end
end
